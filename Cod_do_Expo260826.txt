import React, { useEffect, useState } from 'react';

import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  ActivityIndicator,
  Alert,
  FlatList
} from 'react-native';

import { buscarEndereco } from './services/viacep';
import EnderecoCard from './components/EnderecoCard';
import styles from './style';
import { buscarUsario } from './services/usuario';

export default function App() {
  const [cep, setCep] = useState('');
  const [endereco, setEndereco] = useState(null);
  const [carregando, setCarregando] = useState(false);
  const [usuario, setUsuario] = useState([]);
  const [usuarioSelecionado, setUsuarioSelecionado] = useState({});

  useEffect(() => {
    carregarUsuario();
  }, []);

  async function buscarCep() {
    const cepLimpo = cep.replace(/\D/g, '');

    if (cepLimpo.length !== 8) {
      Alert.alert('CEP inválido', 'Digite um CEP com 8 números.');
      return;
    }

    try {
      setCarregando(true);
      const dados = await buscarEndereco(cepLimpo);

      if (dados.erro) {
        Alert.alert('Erro', 'CEP não encontrado.');
        setEndereco(null);
        return;
      }

      setEndereco(dados);
    } catch (erro) {
      console.log(erro);
      Alert.alert('Erro', 'Não foi possível consultar o CEP.');
    } finally {
      setCarregando(false);
    }
  }

  async function carregarUsuario() {
    try {
      const dados = await buscarUsario();
      setUsuario(dados);
    } catch (error) {
      console.log('Erro:', error);
    }
  }

  

  return (
    <View style={styles.container}>
      <Text style={styles.titulo}>Consulta de CEP</Text>

      <Text style={styles.subtitulo}>
        Digite um CEP para encontrar o endereço
      </Text>

      <FlatList 
        data={usuario}
        keyExtractor={(item) => item.id.toString()}
        renderItem={({item}) => (
          <TouchableOpacity
            style={styles.cadUsuario}
            onPress={() => setUsuarioSelecionado(item)}
          >
           
            <Text style={styles.nomeUsuario}>{item.name}</Text>
            <Text style={styles.emailUsuario}>{item.email}</Text>
            <Text style={styles.textoClique}>Toque para visualizar o endereço</Text>
          </TouchableOpacity>
        )}
      />

      {/* Seus inputs comentados continuam aqui */}
      { /*<TextInput
        style={styles.input}
        placeholder="Digite o CEP"
        keyboardType="numeric"
        maxLength={8}
        value={cep}
        onChangeText={setCep}
      />

      <TouchableOpacity style={styles.botao} onPress={buscarCep}>
        <Text style={styles.textoBotao}>Buscar CEP</Text>
      </TouchableOpacity> */}

    
      {usuarioSelecionado?.address && (
        <View style={styles.card}>
          <Text style={styles.tituloCard}>Endereço</Text>
          <Text>Rua: {usuarioSelecionado.address.street}</Text>
          <Text>Complemento: {usuarioSelecionado.address.suite}</Text>
          <Text>Cidade: {usuarioSelecionado.address.city}</Text>
          <Text>CEP: {usuarioSelecionado.address.zipcode}</Text>
        </View>
      )}
       
      {/* {carregando && <ActivityIndicator size="large" style={styles.loading} />}
      {!carregando && endereco && <EnderecoCard endereco={endereco} />}*/}
    </View>
  );
}
